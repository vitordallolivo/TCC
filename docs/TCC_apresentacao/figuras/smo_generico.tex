    \begin{tikzpicture}[>=latex, font=\small,
        bloco/.style={draw, rectangle, rounded corners=2pt,
                      minimum height=2.6em, align=center, inner sep=5pt},
        soma/.style={draw, circle, inner sep=0pt, minimum size=1.6em}]

        % ── Área sombreada (desenhada antes dos nós) ────────────────────
        \fill[cyan!15, rounded corners=5pt] (0.3,-2.8) rectangle (12.8,2.2);
        \draw[cyan!40, rounded corners=5pt] (0.3,-2.8) rectangle (12.8,2.2);
        \node[font=\scriptsize, text=gray!70, anchor=north] at (5.5,2.15)
            {Observador Não Linear};

        % ── Planta (topo, fora da área sombreada) ───────────────────────
        \node[bloco, minimum width=9em, minimum height=3.8em] (planta) at (5.5,3.5)
            {Planta\\[2pt]\footnotesize$\dot{x}=Ax+Bu$\\[-1pt]\footnotesize$y=Cx$};

        % ── Somador (direita da área sombreada) ─────────────────────────
        \node[soma] (sum) at (11.5,1.0) {};
        \node[font=\tiny] at ($(sum)+(0,0.3)$) {$+$};
        \node[font=\tiny] at ($(sum)+(0,-0.3)$) {$-$};

        % ── Bloco da função de chaveamento ──────────────────────────────
        \node[bloco, minimum width=5em] (sat) at (7.5,1.0) {Sat$(\cdot)$};

        % ── Ganho L ─────────────────────────────────────────────────────
        \node[bloco, minimum width=3em] (L) at (3.5,1.0) {$L$};

        % ── Modelo estimado (base da área sombreada) ─────────────────────
        \node[bloco, minimum width=11em, minimum height=3.8em] (obs) at (7,-1.5)
            {\footnotesize$\dot{\hat{x}}=A\hat{x}+Bu+L\cdot\text{Sat}(y-C\hat{x})$\\[2pt]
             \footnotesize$\hat{y}=C\hat{x}$};
        \node[font=\scriptsize, above=1pt of obs.north] {Modelo Estimado};

        % ── Entrada u ────────────────────────────────────────────────────
        \coordinate (u_jct) at (0.8,2.5);
        \draw[->] (-1.5,2.5) -- node[above]{$u$} (u_jct);
        \fill (u_jct) circle (1.8pt);
        \draw[->] (u_jct) |- (planta.west);

        % u desce pela coluna esquerda até junção com L
        \coordinate (merge_jct) at (1.5,-1.5);
        \draw (u_jct) -- (0.8,-1.5) -- (merge_jct);
        \fill (merge_jct) circle (1.8pt);
        \draw[->] (merge_jct) -- (obs.west);

        % ── L → mesma junção ─────────────────────────────────────────────
        \draw (L.west) -- (1.5,1.0) -- (merge_jct);

        % ── Planta → y ───────────────────────────────────────────────────
        \coordinate (y_fork) at (11.5,3.5);
        \draw (planta.east) -- node[above]{$y$} (y_fork);
        \fill (y_fork) circle (1.8pt);
        \draw[->] (y_fork) -- (sum.north);
        \draw[->] (y_fork) -- ++(1.5,0) node[right]{$y$};

        % ── sum → Sat → L ─────────────────────────────────────────────────
        \draw[->] (sum.west) -- node[above]{\footnotesize$y-C\hat{x}$} (sat.east);
        \draw[->] (sat.west) -- (L.east);

        % ── Modelo estimado → ŷ ───────────────────────────────────────────
        \coordinate (yhat_fork) at (11.5,-1.5);
        \draw (obs.east) -- node[above]{$\hat{y}$} (yhat_fork);
        \fill (yhat_fork) circle (1.8pt);
        \draw[->] (yhat_fork) -- (sum.south);
        \draw[->] (yhat_fork) -- ++(1.5,0) node[right]{$\hat{y}$};

    \end{tikzpicture}
